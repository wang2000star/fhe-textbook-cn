cmake_minimum_required(VERSION 3.16.3)
project(fhe_textbook_meta_bts CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenFHE 1.6.1 CONFIG REQUIRED)
if(NOT BASE_OPENFHE_VERSION VERSION_EQUAL "1.6.1")
    message(FATAL_ERROR "This recorded experiment targets OpenFHE 1.6.1")
endif()
if(NOT OpenFHE_NATIVE_SIZE EQUAL 64)
    message(FATAL_ERROR "This recorded experiment requires NATIVE_SIZE=64")
endif()
set(CMAKE_CXX_FLAGS "${OpenFHE_CXX_FLAGS}")
set(CMAKE_EXE_LINKER_FLAGS "${OpenFHE_EXE_LINKER_FLAGS}")
include_directories(${OPENMP_INCLUDES} ${OpenFHE_INCLUDE}
    ${OpenFHE_INCLUDE}/third-party/include ${OpenFHE_INCLUDE}/core
    ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe)
link_directories(${OpenFHE_LIBDIR} ${OPENMP_LIBRARIES})
if(UNIX AND NOT APPLE)
    add_link_options(-Wl,--no-as-needed)
endif()
add_executable(fhe-textbook-meta-bts openfhe_meta_bts.cpp)
target_link_libraries(fhe-textbook-meta-bts ${OpenFHE_SHARED_LIBRARIES})
